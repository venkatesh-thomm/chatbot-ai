class ServiceNotConfigured(RuntimeError):
    """Raised until the AWS RAG adapters are implemented and configured."""


class RAGService:
    """Application-level interface for chat and document ingestion.

    Phase 1 deliberately does not import the old GCP modules. Importing the
    previous v6_main.py initializes Google clients and downloads utility code
    at import time, which is unsafe and unsuitable for an AWS migration.

    Later phases will implement these methods using:
      * S3 for source documents
      * LangChain RecursiveCharacterTextSplitter
      * Bedrock Titan embeddings
      * OpenSearch Serverless for vector search
      * Amazon Bedrock for answer generation
    """

    def answer(self, *, query: str, max_output_tokens: int, temperature: float,
               top_p: float, top_k: int) -> dict:
        raise ServiceNotConfigured(
            "RAG backend is not connected yet. The FastAPI layer is ready; "
            "AWS Bedrock and OpenSearch adapters are implemented in a later phase."
        )

    def ingest(self, *, bucket_name: str | None, path: str | None,
               repository_url: str | None, branch: str) -> dict:
        raise ServiceNotConfigured(
            "Ingestion backend is not connected yet. S3/GitHub ingestion is implemented in a later phase."
        )


rag_service = RAGService()
